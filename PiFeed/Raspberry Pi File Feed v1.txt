# --- Python backend unchanged ---
def read(f):
    try: return open(os.path.join(UPLOAD,f)).read()
    except: return ""

def fsize(p):
    s=os.path.getsize(p)
    return f"{round(s/1024,1)} KB" if s<1024*1024 else f"{round(s/1024/1024,1)} MB"

def ftime(ts):
    if not ts: return "unknown"
    return time.strftime("%Y-%m-%d %H:%M", time.localtime(ts))

def fname(f): return os.path.splitext(f)[0]
def ftype(f): return os.path.splitext(f)[1][1:].upper() or "FILE"

def stats():
    cpu=psutil.cpu_percent()
    ram=psutil.virtual_memory().percent
    du=psutil.disk_usage('/')
    disk=f"{round(du.used/1e9,1)}/{round(du.total/1e9,1)} GB"
    try: temp=round(psutil.sensors_temperatures()['cpu_thermal'][0].current,1)
    except: temp=0
    uptime=time.strftime('%H:%M:%S',time.gmtime(time.time()-psutil.boot_time()))
    return cpu,ram,disk,temp,uptime

def home():

@app.route("/create",methods=["POST"])
def create():
    name=request.form["name"].strip()
    content=request.form["content"]
    if name:
        fname=name+".txt"
        with open(os.path.join(UPLOAD,fname),"w") as f:
            f.write(content)
        meta.setdefault(fname,{"likes":0,"comments":[],"uploaded":time.time()})
        save(meta)
    return redirect("/")

@app.route("/rename/<old>/<new>")
def rename(old,new):
    oldp=os.path.join(UPLOAD,old)
    newp=os.path.join(UPLOAD,new)
    if os.path.exists(oldp) and not os.path.exists(newp):
        os.rename(oldp,newp)
        meta[new]=meta.pop(old)
        save(meta)
    return redirect("/")

@app.route("/uploads/<f>")
def up(f): return send_from_directory(UPLOAD,f)

@app.route("/delete/<f>")
def d(f):
    os.remove(os.path.join(UPLOAD,f))
    meta.pop(f,None); save(meta)
    return redirect("/")

@app.route("/like/<f>",methods=["POST"])
